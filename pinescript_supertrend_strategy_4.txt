atrPeriod = 11//input(7, "ATR Length")
factor = 1//input.float(1.0, "Factor", step = 0.01)

signals = input.string('Both',options = ['Only Buy','Only Sell','Both'])

//[_, direction] = ta.supertrend(factor, atrPeriod)

[supertrend, direction] = ta.supertrend(factor, atrPeriod)

// fill(bodyMiddle, upTrend, color.new(color.green, 90), fillgaps=false)

buy_alert = input.string('Add Syntax', title = "LONG")
sell_alert = input.string('Add Syntax', title = "SHORT")
exitSyntax = input.string('Add Syntax', title = "LONG EXIT")
exitSyntax2 = input.string('Add Syntax', title = "SHORT EXIT")


if ta.change(direction) < 0 and (signals == 'Both' or signals == 'Only Buy')
    strategy.entry("My Long Entry Id", strategy.long,stop = high,alert_message = buy_alert)

if ta.change(direction) > 0 and (signals == 'Both' or signals == 'Only Sell')
    strategy.entry("My Short Entry Id", strategy.short,stop=low,alert_message=sell_alert)

if signals == 'Only Buy' and (direction) > 0
    strategy.close_all()

if signals == 'Only Sell' and (direction) < 0
    strategy.close_all()

target = input(130.00,'TARGET')
sl = input(90.00,'STOP LOSS')

strategy.exit('long exit','My Long Entry Id',stop = strategy.position_avg_price - sl,limit = strategy.position_avg_price + target,alert_message = exitSyntax)
strategy.exit('short exit','My Short Entry Id',stop =  strategy.position_avg_price + sl,limit = strategy.position_avg_price - target,alert_message = exitSyntax2)

plot(strategy.position_size > 0 ? strategy.position_avg_price + target : na , color = color.rgb(74, 255, 2),style = plot.style_linebr,linewidth = 2)
plot(strategy.position_size < 0 ? strategy.position_avg_price - target : na , color = color.rgb(74, 255, 2),style = plot.style_linebr,linewidth = 2)

plot(strategy.position_size > 0? strategy.position_avg_price - sl : na , color = color.rgb(250, 58, 58),style = plot.style_linebr,linewidth = 2)
plot(strategy.position_size < 0 ? strategy.position_avg_price + sl : na , color = color.rgb(255, 72, 72),style = plot.style_linebr,linewidth = 2)
